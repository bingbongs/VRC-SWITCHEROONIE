cmake_minimum_required(VERSION 3.25)
project(VRC_SWITCHEROONIE VERSION 0.2.5 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The native runtime backend requires Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_library(minhook STATIC third_party/minhook/src/buffer.c third_party/minhook/src/hook.c third_party/minhook/src/trampoline.c third_party/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC third_party/minhook/include)
add_library(switcheroonie_core STATIC native/core.cpp native/ipc.cpp native/config_loader.cpp)
target_include_directories(switcheroonie_core PUBLIC native)
target_include_directories(switcheroonie_core SYSTEM PUBLIC third_party/openvr/headers)
target_link_libraries(switcheroonie_core PUBLIC advapi32 shell32 ole32 bcrypt)
add_library(switcheroonie_hooks STATIC native/hooks.cpp native/known_hook.cpp)
target_include_directories(switcheroonie_hooks PRIVATE third_party/minhook/src/hde)
target_link_libraries(switcheroonie_hooks PUBLIC switcheroonie_core minhook bcrypt)
add_library(driver_switcheroonie SHARED native/driver.cpp)
target_link_libraries(driver_switcheroonie PRIVATE switcheroonie_hooks)
set_target_properties(driver_switcheroonie PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/driver/switcheroonie/bin/win64" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/driver/switcheroonie/bin/win64" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/driver/switcheroonie/bin/win64")
add_executable(switcheroonie-diagnostics native/diagnostics.cpp)
add_executable(switcheroonie-path-probe native/driver_path_probe.cpp)
target_link_libraries(switcheroonie-diagnostics PRIVATE switcheroonie_core "${CMAKE_SOURCE_DIR}/third_party/openvr/lib/win64/openvr_api.lib" bcrypt)
add_custom_command(TARGET switcheroonie-diagnostics POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/third_party/openvr/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:switcheroonie-diagnostics>")
foreach(target switcheroonie_core switcheroonie_hooks driver_switcheroonie switcheroonie-diagnostics)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
  endif()
endforeach()
enable_testing()
add_executable(switcheroonie-native-tests native/tests.cpp)
target_link_libraries(switcheroonie-native-tests PRIVATE switcheroonie_core)
add_test(NAME native-routing COMMAND switcheroonie-native-tests)
add_executable(switcheroonie-startup-tests native/startup_tests.cpp)
target_link_libraries(switcheroonie-startup-tests PRIVATE switcheroonie_core)
if(MSVC)
  target_compile_options(switcheroonie-startup-tests PRIVATE /W4 /permissive- /EHsc)
endif()
add_test(NAME isolated-startup-context COMMAND switcheroonie-startup-tests)
add_executable(switcheroonie-hook-tests native/hook_tests.cpp native/fake_calls.cpp)
target_link_libraries(switcheroonie-hook-tests PRIVATE switcheroonie_hooks)
add_test(NAME isolated-host-hooks COMMAND switcheroonie-hook-tests)
configure_file(native/driver.vrdrivermanifest "${CMAKE_BINARY_DIR}/driver/switcheroonie/driver.vrdrivermanifest" COPYONLY)
configure_file(native/resources/settings/default.vrsettings "${CMAKE_BINARY_DIR}/driver/switcheroonie/resources/settings/default.vrsettings" COPYONLY)
add_test(NAME packaged-driver-resources COMMAND ${CMAKE_COMMAND}
  "-DDRIVER_ROOT=${CMAKE_BINARY_DIR}/driver/switcheroonie"
  -P "${CMAKE_SOURCE_DIR}/native/verify_driver_resources.cmake")
add_subdirectory(tests/test-scene)
add_subdirectory(tests/foreign-spacecal-fixture)
