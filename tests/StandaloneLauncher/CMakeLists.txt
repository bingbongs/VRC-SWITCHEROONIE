cmake_minimum_required(VERSION 3.25)
project(SwitcheroonieLauncherFixture LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Fixture requires Windows x64 MSVC.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(launcher-fixture-payload WIN32 fixture_payload.cpp)
target_compile_definitions(launcher-fixture-payload PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(launcher-fixture-payload PRIVATE /W4 /WX /permissive- /EHsc)
target_link_libraries(launcher-fixture-payload PRIVATE shell32)
