cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(SwitcheroonieTestScene LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  enable_testing()
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The controlled rendering test requires Windows x64.")
endif()
get_filename_component(SWITCHEROONIE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(switcheroonie-test-scene main.cpp)
target_include_directories(switcheroonie-test-scene PRIVATE "${SWITCHEROONIE_ROOT}/third_party/openvr/headers" "${SWITCHEROONIE_ROOT}/native")
if(TARGET switcheroonie_core)
  target_link_libraries(switcheroonie-test-scene PRIVATE switcheroonie_core)
else()
  target_sources(switcheroonie-test-scene PRIVATE "${SWITCHEROONIE_ROOT}/native/ipc.cpp")
endif()
target_link_libraries(switcheroonie-test-scene PRIVATE "${SWITCHEROONIE_ROOT}/third_party/openvr/lib/win64/openvr_api.lib" d3d11 dxgi d3dcompiler advapi32 user32)
set_target_properties(switcheroonie-test-scene PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/Release" RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/Debug")
if(MSVC)
  target_compile_options(switcheroonie-test-scene PRIVATE /W4 /permissive- /EHsc)
endif()
add_custom_command(TARGET switcheroonie-test-scene POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SWITCHEROONIE_ROOT}/third_party/openvr/bin/win64/openvr_api.dll" "$<TARGET_FILE_DIR:switcheroonie-test-scene>")
add_test(NAME controlled-scene-self-test COMMAND switcheroonie-test-scene --self-test)
