cmake_minimum_required(VERSION 3.25)
project(SwitcheroonieStandaloneLauncher LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The standalone launcher requires Windows x64 and MSVC.")
endif()
if(NOT DEFINED SWITCHEROONIE_INNER_SHA256)
  message(FATAL_ERROR "Set SWITCHEROONIE_INNER_SHA256 to the published App/VRC-SWITCHEROONIE.exe SHA256.")
endif()
string(LENGTH "${SWITCHEROONIE_INNER_SHA256}" inner_hash_length)
if(NOT inner_hash_length EQUAL 64 OR NOT SWITCHEROONIE_INNER_SHA256 MATCHES "^[0-9A-Fa-f]+$")
  message(FATAL_ERROR "SWITCHEROONIE_INNER_SHA256 must contain exactly 64 hexadecimal characters.")
endif()
string(TOUPPER "${SWITCHEROONIE_INNER_SHA256}" SWITCHEROONIE_INNER_SHA256)
if(NOT DEFINED SWITCHEROONIE_PAYLOAD_INVENTORY OR NOT EXISTS "${SWITCHEROONIE_PAYLOAD_INVENTORY}")
  message(FATAL_ERROR "Set SWITCHEROONIE_PAYLOAD_INVENTORY to UTF8 SHA256|canonicalRelativePath lines.")
endif()
file(SIZE "${SWITCHEROONIE_PAYLOAD_INVENTORY}" inventory_bytes)
if(inventory_bytes GREATER 1048576 OR inventory_bytes EQUAL 0)
  message(FATAL_ERROR "Payload inventory must be nonempty and at most 1 MiB.")
endif()
file(STRINGS "${SWITCHEROONIE_PAYLOAD_INVENTORY}" inventory_lines ENCODING UTF-8)
set(payload_pins "")
set(seen_paths "")
set(has_inner FALSE)
foreach(line IN LISTS inventory_lines)
  string(FIND "${line}" "|" separator)
  if(NOT separator EQUAL 64)
    message(FATAL_ERROR "Malformed payload inventory line.")
  endif()
  string(SUBSTRING "${line}" 0 64 hash)
  string(SUBSTRING "${line}" 65 -1 path)
  string(LENGTH "${path}" path_length)
  string(FIND "${path}" "\\" backslash)
  string(FIND "${path}" ";" semicolon)
  if(NOT hash MATCHES "^[0-9A-Fa-f]+$" OR path_length LESS 1 OR path_length GREATER 1024 OR
     NOT backslash EQUAL -1 OR NOT semicolon EQUAL -1 OR path MATCHES "[<>:\"|?*\t]" OR
     path MATCHES "^/|/$|//|(^|/)\\.\\.?(/|$)|[. ](/|$)")
    message(FATAL_ERROR "Noncanonical payload inventory path or hash.")
  endif()
  string(TOLOWER "${path}" folded)
  if(folded IN_LIST seen_paths OR folded STREQUAL "package-hashes.json" OR
     folded STREQUAL "portable/vrc-switcheroonie.exe")
    message(FATAL_ERROR "Duplicate or post-wrapper payload inventory path.")
  endif()
  list(APPEND seen_paths "${folded}")
  string(REPLACE "/" ";" components "${path}")
  foreach(component IN LISTS components)
    string(TOUPPER "${component}" component_upper)
    if(component_upper MATCHES "^(CON|PRN|AUX|NUL|COM[1-9]|LPT[1-9]|CONIN\\$|CONOUT\\$)(\\.|$)")
      message(FATAL_ERROR "Reserved Windows device name in payload inventory.")
    endif()
  endforeach()
  string(TOUPPER "${hash}" hash)
  if(path STREQUAL "VRC-SWITCHEROONIE.exe")
    if(NOT hash STREQUAL SWITCHEROONIE_INNER_SHA256)
      message(FATAL_ERROR "Inner launcher inventory hash does not match SWITCHEROONIE_INNER_SHA256.")
    endif()
    set(has_inner TRUE)
  endif()
  string(APPEND payload_pins "    {L\"${path}\", \"${hash}\"},\n")
endforeach()
list(LENGTH seen_paths inventory_count)
if(NOT has_inner OR inventory_count GREATER 4096)
  message(FATAL_ERROR "Inventory must include the exact inner launcher and at most 4096 files.")
endif()
configure_file(launcher_config.hpp.in launcher_config.hpp @ONLY)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(switcheroonie-standalone-launcher WIN32 launcher.cpp launcher.manifest)
set_target_properties(switcheroonie-standalone-launcher PROPERTIES
  OUTPUT_NAME "VRC-SWITCHEROONIE" CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
target_compile_definitions(switcheroonie-standalone-launcher PRIVATE
  WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(switcheroonie-standalone-launcher PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_include_directories(switcheroonie-standalone-launcher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(switcheroonie-standalone-launcher PRIVATE bcrypt shell32 user32)
